# Duration and intonation model. FORMAT: key = number (see languages/hr).
# Spanish is syllable-timed: unstressed vowels keep almost their full
# length, so the contrast between stressed and unstressed syllables is
# smaller than in English, and questions end in a clear rise.
# =============================================================================

pause_colon     = 250

pause_period    = 440

continuation_final = 1.08
